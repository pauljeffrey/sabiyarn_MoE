# One requirements file for everything: training (vast.ai / bare box / Modal images), the eval suite,
# data prep and tests.   pip install -r requirements.txt
#
# Chosen for sabiyarn/model/modeling.py + configuration.py:
#   * transformers 5.x API: `_tied_weights_keys` as a dict, `all_tied_weights_keys`,
#     `mark_tied_weights_as_initialized(loading_info)`, `get_expanded_tied_weights_keys`. The file
#     says "4.x-5.x" but is written against 5.x, and every checkpoint was saved by 5.14.1 (see
#     config.json's transformers_version) -- pinned exactly so a rented box behaves like the run
#     that produced the weights.
#   * torch >= 2.8: FlexAttention (torch.nn.attention.flex_attention, needs >= 2.5 and is far more
#     robust from 2.7 on), scaled_dot_product_attention with explicit masks, bf16 autocast.
#   * huggingface_hub is deliberately NOT pinned: transformers 5 requires 1.x (the old 0.34 pin
#     conflicts), and datasets/tokenizers/safetensors follow from it.
#
# torch is a normal requirement, but on a rented GPU box prefer an image that already ships a CUDA
# build of torch >= 2.8 -- the constraint is then already satisfied and nothing large is downloaded.
torch>=2.8.0
transformers==5.14.1
accelerate>=1.10.0,<2
safetensors>=0.5.0
huggingface_hub
tokenizers
numpy>=1.26
PyYAML>=6.0
omegaconf>=2.3
python-dotenv
structlog>=24.1
tqdm
boto3>=1.34

# experiment tracking (mlflow's system metrics need psutil, and nvidia-ml-py for GPU stats)
mlflow>=2.16
wandb>=0.19
psutil
nvidia-ml-py

# data prep, eval_suite (translation metrics), rl/ (chrF reward) and the legacy eval/eval.py
datasets>=4.0
sacrebleu>=2.4
scikit-learn

# launching runs on Modal from your laptop (`modal run ...`); not needed on a vast.ai box
modal>=1.5

# training.use_cce: fuses lm_head + cross-entropy so the (B, T, 52050) logits tensor is never materialized
# (~2.5 GiB per micro-batch at block_size 4096). Triton kernels, so Linux + an Ampere-or-newer GPU; the import
# is optional and guarded (training/new_train.py logs use_cce_ignored_package_missing if it is absent), and on
# macOS it installs but cannot run. Set training.use_cce: false to skip it entirely.
cut-cross-entropy>=25.1.1; platform_system == "Linux"

# tests: pytest tests/
pytest
